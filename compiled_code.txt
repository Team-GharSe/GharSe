"use client";

import { useRef } from "react";
import {
  gsap,
  prefersReducedMotion,
  ScrollTrigger,
  useGSAP,
} from "@/lib/motion/gsap";
import styles from "./Marquee.module.css";

const ITEMS = [
  "Bakers",
  "Potters",
  "Tailors",
  "Pickle makers",
  "Jewellers",
  "Crafters",
];

export default function Marquee() {
  const track = useRef<HTMLDivElement>(null);

  // The strip scrolls by CSS. Scrolling the page speeds it up and flips its direction.
  useGSAP(
    () => {
      if (prefersReducedMotion() || !track.current) return;
      const anim = track.current.getAnimations()[0];
      if (!anim) return;

      anim.currentTime = 100000; // keeps reverse playback inside an iteration
      const proxy = { rate: 1 };
      const apply = () => {
        anim.playbackRate = proxy.rate;
      };

      ScrollTrigger.create({
        onUpdate: (self) => {
          const boost = 1 + Math.min(Math.abs(self.getVelocity()) / 300, 5);
          gsap.to(proxy, {
            rate: self.direction * boost,
            duration: 0.3,
            overwrite: true,
            onUpdate: apply,
            onComplete: () => {
              gsap.to(proxy, {
                rate: self.direction,
                duration: 1.2,
                onUpdate: apply,
              });
            },
          });
        },
      });
    },
    { scope: track },
  );

  return (
    <div className={styles.marquee} aria-hidden="true">
      <div ref={track} className={styles.track}>
        {[...ITEMS, ...ITEMS].map((item, i) => (
          <span key={i}>{item}</span>
        ))}
      </div>
    </div>
  );
}

    vars: {

      "--w": "90px",
      "--h": "150px",
      "--c": "var(--chilli)",
      "--br": "45px",
    },
  },
];

function renderItem(item: ShelfItem) {
  return (
    <Link
      key={item.label}
      href={item.href}
      className={styles.thing}
      style={item.vars}
    >
      <span className={cx(styles.shape, item.shape && styles[item.shape])} />
      <span className={styles.label}>{item.label}</span>
    </Link>
  );
}

export default function Shelf() {
  const topRow = useRef<HTMLDivElement>(null);
  const bottomRow = useRef<HTMLDivElement>(null);

  useGSAP(() => {
    if (prefersReducedMotion()) return;

    // dir 1: items start off-screen right and slide left. dir -1: start off-screen left and slide right.
    const slideIn = (row: HTMLElement | null, dir: 1 | -1) => {
      if (!row) return;
      const items = gsap.utils.toArray<HTMLElement>(`.${styles.thing}`, row);
      gsap.from(items, {
        x: () => dir * window.innerWidth,
        duration: 1.3,
        ease: "power3.out",
        // the item that travels furthest leads, so the row arrives like a train
        stagger: { each: 0.14, from: dir === 1 ? "start" : "end" },
        scrollTrigger: { trigger: row, start: "top 80%", once: true },
      });
    };

    slideIn(topRow.current, 1);
    slideIn(bottomRow.current, -1);
  });

  return (
    <section className={styles.shelf}>
      <div className="wrap">
        <div className={styles.intro}>
          <h2>Start with what you need</h2>
          <span className={cx("hand", styles.hint)}>
            (tap anything on the shelf)
          </span>
        </div>
        <div ref={topRow} className={styles.row}>
          {TOP.map(renderItem)}
        </div>
        <div ref={bottomRow} className={cx(styles.row, styles.rowEnd)}>
          {BOTTOM.map(renderItem)}
        </div>
      </div>
    </section>
  );
}
